name: Build GoaFX

on:
  push:
  workflow_dispatch:

jobs:
  build:
    strategy:
      fail-fast: false
      matrix:
        os: [windows-latest, macos-latest]
    runs-on: ${{ matrix.os }}
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Configure
        run: cmake -B build -DCMAKE_BUILD_TYPE=Release

      - name: Build
        run: cmake --build build --config Release --parallel

      - name: Upload plugin
        uses: actions/upload-artifact@v4
        with:
          name: GoaFX-${{ matrix.os }}
          path: build/GoaFX_artefacts/Release/VST3
          if-no-files-found: error
